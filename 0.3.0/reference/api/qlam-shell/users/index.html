    def register_cli(
    app: typer.Typer, ctx_provider: Callable[[], AppContext], logger: logging.Logger
) -> None:
    """Register user management commands."""
    resource_app = typer.Typer(help="Manage users.", no_args_is_help=True)

    @resource_app.command("list")
    def list_cmd(
        tenant_id: Annotated[
            UUID | None, typer.Option("--tenant-id", help="Filter by tenant UUID")
        ] = None,
        user_ids: Annotated[
            list[UUID] | None,
            typer.Option("--id", help="Filter by user UUID (repeatable, ORed)"),
        ] = None,
        names: Annotated[
            list[str] | None,
            typer.Option(
                "--name",
                help="Filter by name using contains-style wildcard search (repeatable, ORed)",
            ),
        ] = None,
        emails: Annotated[
            list[str] | None,
            typer.Option("--email", help="Filter by email (repeatable, ORed)"),
        ] = None,
        auth_providers: Annotated[
            list[str] | None,
            typer.Option(
                "--auth-provider",
                help="Filter by auth provider name (repeatable, ORed)",
            ),
        ] = None,
        show_all: Annotated[
            bool, typer.Option("--all", help=USER_TENANT_SHOW_ALL_PAGES_HELP)
        ] = False,
        page: Annotated[int, typer.Option("--page", help=PAGE_INDEX_HELP)] = 0,
        size: Annotated[int | None, typer.Option("--size", help=PAGE_SIZE_HELP)] = None,
        sort: Annotated[
            str,
            typer.Option(
                "--sort",
                help=(
                    "Sort criteria (default: created_date,DESC). "
                    "Supported fields: created_date, last_login."
                ),
            ),
        ] = DEFAULT_USER_SORT_ORDER,
        output: Annotated[
            str | None, typer.Option("-o", "--output", help=OUTPUT_FORMAT_HELP)
        ] = None,
        wide: Annotated[bool, typer.Option("--wide", help=WIDE_TABLE_HELP)] = False,
    ) -> None:
        """List users."""
        ctx = ctx_provider()
        logger.debug(
            "Listing users: tenant_id=%s ids=%s names=%s emails=%s auth_providers=%s "
            "all=%s page=%s size=%s sort=%s",
            tenant_id,
            user_ids,
            names,
            emails,
            auth_providers,
            show_all,
            page,
            size,
            sort,
        )

        client = UsersClient(ctx)
        list_kwargs = _users_list_filter_kwargs(
            sort=sort,
            tenant_id=tenant_id,
            user_ids=user_ids,
            names=names,
            emails=emails,
            auth_providers=auth_providers,
        )
        if show_all:
            payload = collect_pages_payload(
                client.iter_pages(size=DEFAULT_ALL_PAGE_SIZE, **list_kwargs)
            )
        else:
            payload = page_to_payload(
                client.list_page(page=page, size=size, **list_kwargs)
            )

        render_with_optional_schema(
            payload,
            _USERS_LIST_SCHEMA,
            ctx=ctx,
            output=output,
            wide=wide,
        )

    @resource_app.command("get")
    def get_cmd(
        user_id: Annotated[UUID, typer.Argument(help="User ID")],
        output: Annotated[
            str | None, typer.Option("-o", "--output", help=OUTPUT_FORMAT_HELP)
        ] = None,
    ) -> None:
        """Get a user by UUID."""
        ctx = ctx_provider()
        logger.debug("Getting user: id=%s", user_id)

        result = UsersClient(ctx).get(id=user_id)
        render(serialize_model(result), output or ctx.effective_output_format, wrap=ctx.wrap)

    @resource_app.command("create")
    def create_cmd(
        data: Annotated[
            str,
            typer.Argument(help="User data as JSON, @file.json, file://path, or '-' for stdin"),
        ],
        output: Annotated[
            str | None, typer.Option("-o", "--output", help=OUTPUT_FORMAT_HELP)
        ] = None,
    ) -> None:
        """Create a user."""
        ctx = ctx_provider()
        logger.debug("Creating user")

        payload = load_json_like(data)
        model = validate_model_payload(payload, CreateUserRequest, field_name="user body")
        result = UsersClient(ctx).create(body=model)
        render(serialize_model(result), output or ctx.effective_output_format, wrap=ctx.wrap)

    @resource_app.command("update")
    def update_cmd(
        user_id: Annotated[UUID, typer.Argument(help="User ID")],
        data: Annotated[
            str,
            typer.Argument(
                help="Updated user data as JSON, @file.json, file://path, or '-' for stdin"
            ),
        ],
        output: Annotated[
            str | None, typer.Option("-o", "--output", help=OUTPUT_FORMAT_HELP)
        ] = None,
    ) -> None:
        """Patch a user's profile."""
        ctx = ctx_provider()
        logger.debug("Updating user: id=%s", user_id)

        payload = load_json_like(data)
        model = validate_model_payload(payload, PatchUserRequest, field_name="user patch body")
        result = UsersClient(ctx).update(id=user_id, body=model)
        render(serialize_model(result), output or ctx.effective_output_format, wrap=ctx.wrap)

    @resource_app.command("details")
    def details_cmd(
        user_id: Annotated[UUID, typer.Argument(help="User ID")],
        output: Annotated[
            str | None, typer.Option("-o", "--output", help=OUTPUT_FORMAT_HELP)
        ] = None,
    ) -> None:
        """Get a detailed user profile."""
        ctx = ctx_provider()
        logger.debug("Getting user details: id=%s", user_id)

        result = UsersClient(ctx).get_details(id=user_id)
        render(serialize_model(result), output or ctx.effective_output_format, wrap=ctx.wrap)

    @resource_app.command("roles")
    def roles_cmd(
        user_id: Annotated[UUID, typer.Argument(help="User ID")],
        output: Annotated[
            str | None, typer.Option("-o", "--output", help=OUTPUT_FORMAT_HELP)
        ] = None,
        wide: Annotated[bool, typer.Option("--wide", help=WIDE_TABLE_HELP)] = False,
    ) -> None:
        """Get a user's role assignments."""
        ctx = ctx_provider()
        logger.debug("Getting user roles: id=%s", user_id)

        _render_role_assignments(
            UsersClient(ctx).get_roles(id=user_id),
            ctx=ctx,
            output=output,
            wide=wide,
        )

    @resource_app.command("set-roles")
    def set_roles_cmd(
        user_id: Annotated[UUID, typer.Argument(help="User ID")],
        data: Annotated[
            str,
            typer.Argument(
                help="Role assignments as JSON array, @file.json, file://path, or '-' for stdin"
            ),
        ],
        output: Annotated[
            str | None, typer.Option("-o", "--output", help=OUTPUT_FORMAT_HELP)
        ] = None,
        wide: Annotated[bool, typer.Option("--wide", help=WIDE_TABLE_HELP)] = False,
    ) -> None:
        """Replace a user's role assignments."""
        ctx = ctx_provider()
        logger.debug("Setting user roles: id=%s", user_id)

        payload = load_json_like(data)
        roles = validate_adapter_payload(payload, ROLE_ASSIGNMENTS_ADAPTER, field_name="roles")
        result = UsersClient(ctx).set_roles(id=user_id, roles=roles)
        _render_role_assignments(
            result,
            ctx=ctx,
            output=output,
            wide=wide,
        )

    @resource_app.command("change-password")
    def change_password_cmd(
        user_id: Annotated[UUID, typer.Argument(help="User ID")],
    ) -> None:
        """Trigger a password-change flow for the user."""
        ctx = ctx_provider()
        logger.debug("Changing user password: id=%s", user_id)

        UsersClient(ctx).change_password(id=user_id)
        typer.echo(f"Password change request sent for user {user_id}.")

    app.add_typer(resource_app, name="users")
    logger.debug("Registered users CLI commands")
